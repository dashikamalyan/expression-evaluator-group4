EXPRESSION TOKENIZER # ============================================================ OPERATORS = {"+", "-", "*", "/", "^"} PRECEDENCE = { "+": 1, "-": 1, "*": 2, "/": 2, "u+": 3, "u-": 3, "^": 4 } RIGHT_ASSOC = {"^", "u+", "u-"} def tokenize(expression): expression = expression.replace(" ", "") if not expression: raise ValueError("Please enter an expression.") pattern = r"\d+(?:\.\d+)?|[()+\-*/^]" tokens = re.findall(pattern, expression) if "".join(tokens) != expression: raise ValueError("Invalid characters in expression.") result = [] previous = None for token in tokens: if token in {"+", "-"} and ( previous is None or previous in OPERATORS or previous in {"(", "u+", "u-"} ): token = "u" + token result.append(token) previous = token balance = 0 expecting_operand = True for token in result: if re.fullmatch(r"\d+(?:\.\d+)?", token): if not expecting_operand: raise ValueError("Missing operator between numbers.") expecting_operand = False elif token == "(": if not expecting_operand: raise ValueError("Missing operator before parentheses.") balance += 1 expecting_operand = True elif token == ")": if expecting_operand: raise ValueError("Invalid or empty parentheses.") balance -= 1 if balance < 0: raise ValueError("Mismatched parentheses.") expecting_operand = False elif token in {"u+", "u-"}: if not expecting_operand: raise ValueError("Unexpected unary operator.") elif token in OPERATORS: if expecting_operand: raise ValueError("Operator is missing an operand.") expecting_operand = True if balance != 0: raise ValueError("Mismatched parentheses.") if expecting_operand: raise ValueError("Expression cannot end with an operator.") return result # ============================================================ # 2. INFIX TO POSTFIX CONVERSION # ============================================================ def infix_to_postfix(expression): input_queue = deque(tokenize(expression)) operator_stack = [] postfix = [] steps = [] while input_queue: token = input_queue.popleft() if re.fullmatch(r"\d+(?:\.\d+)?", token): postfix.append(token) action = "Add operand to postfix queue." elif token == "(": operator_stack.append(token) action = "Push opening bracket into stack." elif token == ")": while operator_stack and operator_stack[-1] != "(": postfix.append(operator_stack.pop()) if not operator_stack: raise ValueError("Mismatched parentheses.") operator_stack.pop() action = "Pop operators until opening bracket." else: while operator_stack and operator_stack[-1] != "(": top = operator_stack[-1] higher = PRECEDENCE[top] > PRECEDENCE[token] equal_left = ( PRECEDENCE[top] == PRECEDENCE[token] and token not in RIGHT_ASSOC ) if higher or equal_left: postfix.append(operator_stack.pop()) else: break operator_stack.append(token) action = "Push operator into stack." steps.append({ "token": token, "queue": list(input_queue), "stack": operator_stack.copy(), "postfix": postfix.copy(), "action": action }) while operator_stack: op = operator_stack.pop() if op == "(": raise ValueError("Mismatched parentheses.") postfix.append(op) steps.append({ "token": "END", "queue": [], "stack": operator_stack.copy(), "postfix": postfix.copy(), "action": "Pop remaining operator." }) return postfix, steps # ============================================================ # 3. POSTFIX EVALUATION # ============================================================ def evaluate_postfix(postfix): stack = [] steps = [] for token in postfix: if re.fullmatch(r"\d+(?:\.\d+)?", token): stack.append(float(token)) action = f"Push {token} into stack." elif token in {"u+", "u-"}: if len(stack) < 1: raise ValueError("Missing operand.") a = stack.pop() result = -a if token == "u-" else a stack.append(result) action = f"Apply {token} to {a:g}." elif token in OPERATORS: if len(stack) < 2: raise ValueError("Invalid postfix expression.") b = stack.pop() a = stack.pop() if token == "+": result = a + b elif token == "-": result = a - b elif token == "*": result = a * b elif token == "/": if b == 0: raise ZeroDivisionError("Cannot divide by zero.") result = a / b elif token == "^": if abs(b) > 10000: raise ValueError("Exponent is too large.") result = a ** b if isinstance(result, complex) or not math.isfinite(result): raise ValueError("Result outside supported numeric range.") stack.append(result) action = f"{a:g} {token} {b:g} = {result:g}" steps.append({ "token": token, "stack": stack.copy(), "action": action }) if len(stack) != 1: raise ValueError("Invalid expression.") return stack[0], steps # ============================================================ # 4. EXPRESSION TREE # ============================================================ def create_expression_tree(postfix): graph = nx.DiGraph() node_stack = [] labels = {} count = 0 for token in postfix: node = count count += 1 graph.add_node(node) labels[node] = token if token in OPERATORS: if len(node_stack) < 2: raise ValueError("Invalid expression tree.") right = node_stack.pop() left = node_stack.pop() graph.add_edge(node, left) graph.add_edge(node, right) elif token in {"u-", "u+"}: if not node_stack: raise ValueError("Invalid expression tree.") child = node_stack.pop() graph.add_edge(node, child) node_stack.append(node) if len(node_stack) != 1: raise ValueError("Invalid expression tree.") positions = {} leaf_counter = [0] def arrange(node, depth=0): children = list(graph.successors(node)) if not children: x = leaf_counter[0] leaf_counter[0] += 1 else: child_x = [ arrange(child, depth + 1) for child in children ] x = sum(child_x) / len(child_x) positions[node] = (x, -depth) return x arrange(node_stack[0]) return graph, labels, positions # ============================================================ # 5.
